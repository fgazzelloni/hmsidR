\usepackage{xcolor}
\usepackage{graphicx}
\usepackage{etoolbox}

% Command to convert images to grayscale
\makeatletter
\newcommand{\set@grayscale}{%
  \AtBeginDocument{%
    \let\oldincludegraphics\includegraphics
    \renewcommand{\includegraphics}[2][]{%
      \begingroup
      \oldincludegraphics[##1]{##2}%
      \endgroup
    }%
    \let\oldsetkeys\setkeys
    \renewcommand{\setkeys}[2][]{%
      \begingroup
      \def\KV@Gin@bb{%
        \ifGin@pdftex
          \Gin@temp
          \pdfxform 0\relax
          \dp 0pt\ht \Gin@temp \dp \Gin@temp \ht
          \expandafter\expandafter\expandafter\pgfsys@gbox@sizespec@gin
          \expandafter\expandafter\expandafter{%
            \expandafter\Gin@bb@spec\Gin@bb
          }%
        \fi
      }%
      \oldsetkeys{##1}{##2}%
      \endgroup
    }%
  }%
}
\makeatother

\set@grayscale
